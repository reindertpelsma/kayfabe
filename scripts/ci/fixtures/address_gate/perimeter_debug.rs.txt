// G1b known positives: a derived Debug on a perimeter type that holds an address. The header line
// of each offending item carries the marker the self-test expects a hit at.
#[derive(Debug)]
pub struct Mapping { // HIT
    base: NonNull<u8>,
    len: usize,
}

#[derive(Debug, Clone, Copy)]
struct Raw(*mut u8, usize); // HIT

#[derive(Debug)]
enum Hook { // HIT
    Some(*const c_void),
    None,
}

#[derive(Clone, Debug)]
#[repr(C)]
pub(crate) struct Handle { // HIT
    raw: usize,
}

// widened 2026-10-04 (review of v3-sec-rawaddr): driver handles, the device-address alias, and a

}

struct DevRange { // HIT
    addr: DevAddr,
    len: u64,
}

#[derive(Debug)]
struct ImportHandle { // HIT
    ctx: Ctx,
    handle: u64,
}

#[derive(Debug, Clone, Copy, Default)]
struct UserspaceMemoryRegion { // HIT
    slot: u32,
    guest_phys_addr: u64,
    userspace_addr: u64,
}
