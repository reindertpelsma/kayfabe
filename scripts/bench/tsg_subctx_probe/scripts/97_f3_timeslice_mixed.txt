# TSG-level controls on a multi-subctx TSG: SET_TIMESLICE twice, schedule enable=0 then enable=1 around work
tsg engine=gr vas=0
ctx S flags=sync vas=0
ctx A flags=async vas=0
chan g ctx=S engine=gr objs=3d+cmp
chan c ctx=A engine=gr rawflags=0x10 objs=cmp
bindtsg
timeslice us=500
schedtsg
work g,c
timeslice us=2000
work g,c
schedtsg enable=0
work g,c wait=800
schedtsg enable=1
work g,c
