# FREE ORDER: free a ctxshare while a channel still uses it; then free channel then ctxshare; VEID reuse; TSG keeps running
tsg engine=gr vas=0
ctx S flags=sync vas=0
ctx A flags=async vas=0
chan g ctx=S engine=gr objs=3d+cmp
chan c ctx=A engine=gr rawflags=0x10 objs=cmp
bindtsg
schedtsg
work g,c
freectx A
work g,c
freech c
freectx A
ctx A2 flags=async vas=0
chan c2 ctx=A2 engine=gr rawflags=0x10 objs=cmp
work g,c2
freech c2
freectx A2
ctx A3 flags=async vas=0
work g
freetsg
