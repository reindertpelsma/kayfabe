# CONTROL SCOPE: DISABLE_CHANNELS (bOnlyDisableScheduling=0) on ONE channel of a 3-subcontext TSG, then enable
tsg engine=gr vas=0
ctx S flags=sync vas=0
ctx A flags=async vas=0
ctx B flags=async vas=0
chan g ctx=S engine=gr objs=3d+cmp
chan c ctx=A engine=gr rawflags=0x10 objs=cmp
chan k ctx=B engine=gr rawflags=0x10 objs=ce
bindtsg
schedtsg
work g,c,k
disable c
work g,k wait=1500
work c wait=1500
enable c
work g,c,k wait=1500
