# CONTROL SCOPE: TSG-level schedule(bEnable=0) with longer waits, then enable; and DISABLE_CHANNELS naming every channel
tsg engine=gr vas=0
ctx S flags=sync vas=0
ctx A flags=async vas=0
chan g ctx=S engine=gr objs=3d+cmp
chan c ctx=A engine=gr rawflags=0x10 objs=cmp
bindtsg
schedtsg
work g,c
schedtsg enable=0
settle 300
work g,c wait=2000
schedtsg enable=1
work g,c wait=2000
disable g,c
work g,c wait=1500
enable g,c
work g,c wait=1500
