# Vulkan-like: SYNC ctxshare graphics channel (3D+cmp+ce) + ASYNC compute channel (rq1) in ONE TSG, graphics and compute run concurrently
tsg engine=gr vas=0
ctx S flags=sync vas=0
ctx A flags=async vas=0
chan g ctx=S engine=gr objs=ce+3d+cmp
chan c ctx=A engine=gr rawflags=0x10 objs=cmp
bindtsg
schedtsg
work g,c
bench g,c iters=500
