def rehearse_adapter(
    *, bad_source=False, missing_source=False, bad_capture=False, native_refusal=False, cpus=8
):
    with tempfile.TemporaryDirectory(prefix="sw5-full-rehearsal-") as tmp:
        root = Path(tmp)
        kit = root / "kit"
        kit.mkdir()
        files = [
            "ready.py",
            "pull.py",
            "collect_capture.py",
            "console_pin.py",
            "provider_retry.py",
            "readiness.py",
            "round_budget.py",
            "precreate_watchdog.py",
            "run_box.sh",
            "driver_inventory.sh",
            "guard_hooks.sh",
            "die_binding.py",
            "prepare_binding.py",
            "final_xids.py",
            "quiesce.py",
            "lease_guard.py",
            "final_capture.py",
            "network_guard.py",
            "pending_die.py",
            "completed.json",
            "framebuffer.json",
        ]
        for filename in files:
            shutil.copyfile(O.HERE / filename, kit / filename)
        for filename in ("instruments.tar", "derived.tar"):
            # Small source-locked text-only kit fixture, never a production payload.
            (kit / filename).write_bytes(b"fake source kit fixture")
        locks = {str(p): hashlib.sha256(p.read_bytes()).hexdigest() for p in kit.iterdir()}
        if bad_source:
            (kit / "run_box.sh").write_text("changed")
        if missing_source:
            (kit / "run_box.sh").unlink()
        try:
            lifecycle.source_lock(locks)
        except FileNotFoundError as error:
            raise ValueError("missing source lock") from error
        state = {
            "key": KEY,
            "present": False,
            "bad_capture": bad_capture,
            "native_refusal": native_refusal,
            "cpus": cpus,
        }
        (root / "state.json").write_text(json.dumps(state))
        with tarfile.open(root / "capture.tgz", "w:gz") as archive:
            data = b"private text fixture evidence\n"
            member = tarfile.TarInfo("evidence/controller.log")
            member.size = len(data)
            archive.addfile(member, io.BytesIO(data))
        binary = root / "bin"
        binary.mkdir()
        for name in ("vastai", "ssh", "scp", "ssh-keyscan"):
            path = binary / name
            path.write_text("#!" + sys.executable + "\n" + FAKE)
            path.chmod(0o700)
        report = root / "report.md"
        now = dt.datetime.now(dt.timezone.utc)
        plan = B.plan(0.34, 23, now, transfer_usd=0.2)
        env = {"PATH": str(binary) + ":" + os.environ["PATH"], "SW5_FAKE_ROOT": str(root)}
        with patch.dict(os.environ, env):
            backend = O.Backend(
                {"id": 57427095, "machine_id": 99805, "host_id": 70221, "dph_total": 0.34},
                plan,
                23,
                evidence=root / "lease",
                report=report,
                here=kit,
            )
            try:
                result = lifecycle.execute(backend, plan)
            except (ValueError, __import__("subprocess").CalledProcessError):
                if native_refusal or cpus < 8:
                    result = {"refused": True}
                else:
                    raise
            finally:
                # Every fake timer is an actual owned process; do not leak it on test errors.
                if backend.watch is not None and backend.watch.poll() is None:
                    backend.watch.terminate()
                    backend.watch.wait(timeout=5)
        text = "\n".join(
            p.read_text(errors="replace")
            for p in (root / "lease").iterdir()
            if p.is_file() and p.suffix in (".json", ".log", ".md")
        )
        if "FAKE_SECRET_MUST_NOT_PERSIST" in text:
            raise ValueError("raw create secret persisted")
        calls = [json.loads(line) for line in (root / "events.jsonl").read_text().splitlines()]
        events = ["source-lock", "arm"]
        for call in calls:
            args = call["args"]
            if call["tool"] == "vastai" and args[:2] == ["create", "instance"]:
                events.append("create")
            if call["tool"] == "vastai" and args[:2] == ["destroy", "instance"]:
                events.append("destroy")
            if call["tool"] == "ssh" and args[-1].startswith("if test -f"):
                events.extend(["native-binding", "gates9+10-user", "bare30"])
                if not native_refusal:
                    events.extend(
                        ["kf3-build", "host-fastguest-build", "guest30-180s", "xid-user-census"]
                    )
        return result, events
