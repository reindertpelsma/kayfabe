#!/bin/bash
# cand1b_apps.sh — the V3 app matrix at 0ac157b2 (shape of cand1_apps.sh / V3_CDP.md §5.4): host
# baseline, then the guest on guest_apps.qcow2 — ONE boot, no PM (APPS_PER_BOOT=100), then every
# non-PASS app alone. kf3 = the merge bar's own binary (kf3-bins/0ac157b2). No re-provisioning: the
# bundle and guest_apps.qcow2 are the ones cand1 built (bundle sha fb64de820389b448).
L=/root/prov/cand1b_apps2.log
exec >>"$L" 2>&1
echo "JOBSTART $(date -Is)"
trap 'rc=$?; echo "JOBEND rc=$rc $(date -Is)"' EXIT
until grep -aq '^JOBEND rc=' /root/prov/cand1b_mb2job.log 2>/dev/null; do sleep 30; done
echo "MB_DONE_SEEN [$(grep -a '^JOBEND rc=' /root/prov/cand1b_mb2job.log)]"
while pgrep -x qemu-system-x86 >/dev/null; do sleep 10; done
cd /root/kayfabe || exit 90
W=0ac157b2fff15294073b95acbe7377a6133ae5aa
[ "$(git rev-parse HEAD)" = $W ] || { echo "wrong HEAD"; exit 91; }
echo "REV $(git rev-parse HEAD) dirty=$(git status --porcelain --untracked-files=no | wc -l)"
KB=/workspace/bench/kf3-bins/0ac157b2/qemu-system-x86_64
[ -x "$KB" ] || { echo "no kf3 binary $KB"; exit 92; }
echo "KF3_BIN $KB $(stat -c '%s %y' $KB) sha256=$(sha256sum $KB | cut -c1-16)"
[ -s /workspace/bench/guest_apps.qcow2 ] || { echo "no guest_apps.qcow2"; exit 93; }
echo "GUEST_APPS_IMG $(stat -c '%s %y' /workspace/bench/guest_apps.qcow2)"
echo "BUNDLE bundle.tgz sha=$(sha256sum /opt/apps/bundle.tgz | cut -c1-16) $(stat -c '%s %y' /opt/apps/bundle.tgz)"
[ ! -e /workspace/apps/results/cand1b ] || { echo "results/cand1b exists"; exit 94; }
t0=$(date '+%Y-%m-%d %H:%M:%S')
bash scripts/apps/apps_matrix.sh host cand1b all > /root/prov/cand1b_apps_host.log 2>&1; echo "HOST_RC=$? $(date -Is)"
t1=$(date '+%Y-%m-%d %H:%M:%S')
grep -a 'HOST_DONE' /root/prov/cand1b_apps_host.log
echo "HOST_WINDOW [$t0 .. $t1] host_xid_journal=$(journalctl -k --since "$t0" --until "$t1" --no-pager | grep -ac 'Xid')"
KF3_BIN=$KB KF_GUEST_IMG=/workspace/bench/guest_apps.qcow2 APPS_PER_BOOT=100 \
  bash scripts/apps/apps_matrix.sh guest cand1b all > /root/prov/cand1b_apps_guest.log 2>&1; echo "GUEST_RC=$? $(date -Is)"
t2=$(date '+%Y-%m-%d %H:%M:%S')
grep -a 'GUEST_DONE\|kf3 binary\|boot ap_' /root/prov/cand1b_apps_guest.log | cut -c1-240
echo "GUEST_WINDOW [$t1 .. $t2] host_xid_journal=$(journalctl -k --since "$t1" --until "$t2" --no-pager | grep -ac 'Xid')"
R=/workspace/apps/results/cand1b
journalctl -k --since "$t0" --until "$t2" --no-pager | grep -a 'Xid\|NVRM' > $R/host_xid_journal.log
python3 scripts/apps/triage.py $R > $R/triage.txt 2>&1
[ -f $R/guest_isolated.res ] && python3 scripts/apps/triage.py $R/iso guest.res > $R/triage_isolated.txt 2>&1
python3 scripts/apps/summarize.py $R > $R/summary.md 2>&1
grep -a -o -h 'kf3-bin-rev:[0-9a-f]*' $R/boot_*.driver.log $R/iso/boot_*.driver.log 2>/dev/null | sort | uniq -c
echo "APPS_DONE"
